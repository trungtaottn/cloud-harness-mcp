# the scope tool list needs must be named explicitly (the pilot already lost
# docker-cli to --no-install-recommends once).

# --- guest network ownership (required units) ---
# Separate Debian binary packages — NOT pulled in by --no-install-recommends
# unless named. provision.sh fails the build if their unit files are absent:
# resolved generates the /run/systemd/resolve/stub-resolv.conf that
# /etc/resolv.conf points at, networkd renders the per-VM .network the
# supervisor pushes, timesyncd keeps guest clocks sane.
systemd-networkd
systemd-resolved
systemd-timesyncd

# --- guest Docker + Compose (scope) ---
# docker-cli, docker-buildx and docker-compose-v2 are Recommends of
# docker.io, not Depends — they must be listed explicitly.
docker.io
docker-cli
docker-buildx
docker-compose-v2

# --- database clients (scope: PostgreSQL client libs) ---
postgresql-client
libpq-dev
